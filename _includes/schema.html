{%- comment -%} Structured data for search engines: one JSON-LD graph per page, its parts linked by @id. Every page describes the site and its author. Guides add the article and its breadcrumb, plus FAQPage when their front matter has faq and Review when it has review (the score comes from review_scores in _config.yml, as on the page). Other pages pick their type with schema_type in front matter (ProfilePage on About, CollectionPage on Guides); the rest are plain WebPages. default.html drops the JSON-LD that jekyll-seo-tag writes, so the two never disagree. {%- endcomment -%} {%- assign home_url = '/' | absolute_url -%} {%- assign about_url = '/about/' | absolute_url -%} {%- assign guides_url = '/guides/' | absolute_url -%} {%- assign page_url = page.url | absolute_url -%} {%- assign website_id = home_url | append: '#website' -%} {%- assign person_id = about_url | append: '#person' -%} {%- assign webpage_id = page_url | append: '#webpage' -%} {%- assign language = site.lang | default: 'en' -%} {%- assign page_name = page.title | default: site.title -%} {%- assign page_type = page.schema_type | default: 'WebPage' -%} {%- assign is_home = false -%}{%- if page.url == '/' -%}{%- assign is_home = true -%}{%- endif -%} {%- assign is_guide = false -%}{%- if page.collection == 'posts' -%}{%- assign is_guide = true -%}{%- endif -%} {%- assign person_image = site.author.avatar | default: site.author.photo -%} {%- assign same_as = site.author.same_as | default: '' -%} {%- if same_as == '' -%}{%- assign same_as = '' | split: '' -%}{%- endif -%} {%- if site.author.linkedin -%}{%- assign same_as = same_as | unshift: site.author.linkedin -%}{%- endif -%} {%- unless site.substack_url == nil or site.substack_url == '' or site.substack_url contains 'YOUR-NAME' -%}{%- assign same_as = same_as | push: site.substack_url -%}{%- endunless -%} {%- if is_guide -%} {%- assign updated = page.updated | default: page.date -%} {%- assign published_iso = page.date | date_to_xmlschema -%} {%- assign updated_iso = updated | date_to_xmlschema -%} {%- if page.review -%} {%- assign review_score = page.review.rating | default: site.review_scores[page.review.verdict] -%} {%- assign review_type = page.review.type | default: 'Book' -%} {%- endif -%} {%- endif -%}
